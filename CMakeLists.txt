cmake_minimum_required(VERSION 3.22)
project(edge_test_target LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ENABLE_COVERAGE "Enable GCC coverage" OFF)
option(BUILD_TESTING "Build Google Test tests" ON)

find_package(Threads REQUIRED)
add_library(target_core src/target/timing.cpp)
target_include_directories(target_core PUBLIC src/target)
target_compile_options(target_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(target_core PUBLIC Threads::Threads)
add_executable(target_sim src/target/main.cpp)
target_link_libraries(target_sim PRIVATE target_core)
target_compile_options(target_sim PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(ENABLE_COVERAGE)
  target_compile_options(target_core PUBLIC --coverage -O0 -g)
  target_link_options(target_core PUBLIC --coverage)
endif()

if(BUILD_TESTING)
  include(FetchContent)
  FetchContent_Declare(googletest
    URL https://github.com/google/googletest/archive/refs/tags/v1.17.0.tar.gz
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(googletest)
  enable_testing()
  add_executable(target_tests tests/cpp/timing_test.cpp)
  target_link_libraries(target_tests PRIVATE target_core GTest::gtest_main)
  target_compile_options(target_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  include(GoogleTest)
  gtest_discover_tests(target_tests)
endif()
